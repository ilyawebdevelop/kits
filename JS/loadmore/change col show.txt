$(document).ready(function () {
  const $catalogWrap = $('.catalog-wrap');
  const $textMoreButton = $('.text-more');

  // Получаем количество карточек для изначального показа
  const initialCardsToShow = parseInt($catalogWrap.data('initial-show')) || 2;
  // Получаем количество карточек для показа по кнопке "Показать еще"
  const moreCardsToShow = parseInt($catalogWrap.data('show-more')) || 2;

  let cardsVisible = 0; // Счетчик видимых карточек

  const $allCols = $catalogWrap.find('.col');

  // Изначально показываем первые `initialCardsToShow` карточек
  if ($allCols.length > 0) {
    $allCols.slice(0, initialCardsToShow).addClass('visible');
    cardsVisible = initialCardsToShow;
  }

  // Скрываем кнопку, если карточек не больше, чем показывается изначально
  if ($allCols.length <= initialCardsToShow) {
    $textMoreButton.hide();
  }

  // Обработчик клика по кнопке "Показать еще"
  $textMoreButton.on('click', function () {
    let cardsToShowThisClick = moreCardsToShow; // Сколько показать в этот клик
    let $hiddenCards = $catalogWrap.find('.col:not(.visible)'); // Находим скрытые карточки

    if ($hiddenCards.length === 0) {
      // Если скрытых карточек больше нет, скрываем кнопку
      $(this).hide();
      return;
    }

    // Показываем следующие `cardsToShowThisClick` карточек
    $hiddenCards.slice(0, cardsToShowThisClick).each(function () {
      $(this).addClass('visible');
      cardsVisible++; // Увеличиваем счетчик видимых карточек
    });

    // Если после показа всех карточек, их общее количество стало равно всем карточкам,
    // то скрываем кнопку.
    if ($allCols.length === cardsVisible) {
      $(this).hide();
    }
  });
});
