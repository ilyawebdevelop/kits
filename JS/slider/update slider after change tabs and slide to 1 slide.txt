const swiperInstances = [];
// Инициализация слайдера productCatSlider
document.querySelectorAll('.productCatSlider').forEach(n => {

  const productCatSlider = new Swiper(n, {
    slidesPerView: 3,
    spaceBetween: 19,
    speed: 600,
    observer: true,
    navigation: {
      prevEl: n.closest('.sliderW').querySelector('.navArrowPrev'),
      nextEl: n.closest('.sliderW').querySelector('.navArrowNext'),
    },
    scrollbar: {
      el: n.closest('.sliderW').querySelector('.swiper-scrollbar'),
      draggable: true,
    },
    breakpoints: {
      0: {
        slidesPerView: 1,
      },
      576: {
        slidesPerView: 2,
      },
      1400: {
        slidesPerView: 3,
      },
    },
  });
  swiperInstances.push(productCatSlider);
});
console.log(swiperInstances);

UIkit.util.on('.uk-switcher', 'beforeshow', function () {
  console.log('УРА!!!');

  setTimeout(function () {
    swiperInstances.forEach(el => {
      el.update();
      el.slideTo(0);
    });
  }, 500)



  //  console.log(productCatSlider);
});
