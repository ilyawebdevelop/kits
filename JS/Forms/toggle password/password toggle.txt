 <div class="inputW mb-2 position-relative">
        <div class="h5 mb-2">Пароль</div>
        <input type="password" class="formInput formInput--password border-grey-2">
        <div class="togglePasswordBtn"><img src="@img/icons/eye.svg" alt="icon"></div>
      </div>


#togglePassword {
  position: absolute;
  right: 20px;
  top: 17px;
  &:hover {
    opacity: .8;
  }
}

const togglePassword = document.querySelectorAll('.togglePasswordBtn');
togglePassword.forEach(el => {
  let password = el.closest('.inputW').querySelector('.formInput--password');
  el.addEventListener('click', () => {
    // Toggle the type attribute using
    // getAttribure() method
    const type = password.getAttribute('type') === 'password' ?
      'text' : 'password';
    password.setAttribute('type', type);
  });
});
